82def build_system_prompt(assistant_name: str = "Jarvis") -> str: 

83 """Render the persona prompt with the configured assistant name. 

84 

85 The name comes from the user's wake word (capitalised); defaults to 

86 "Jarvis" when no config is available (tests, eval harnesses).

88 name = (assistant_name or "Jarvis").strip() or "Jarvis" 

89 return _SYSTEM_PROMPT_TEMPLATE.format(name=name)
