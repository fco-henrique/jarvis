20class ModelSize(Enum): 

21 """Classification of model sizes for prompt selection.""" 

22 SMALL = "small" # 0.5B-7.5B - needs explicit tool constraints 

23 LARGE = "large" # 8B+ - can infer tool usage from context 

24 

25 

26# Threshold in billions of parameters — models at or below this are SMALL. 

27# Set at 7.5 (midpoint between 7B and 8B) so 7B models are SMALL and 8B+ are LARGE. 

28_SMALL_MODEL_THRESHOLD = 7.5 

29 

30# Model families whose bare names (no size tag) should be treated as SMALL. 

31# Sized variants follow the size threshold like any other model — e.g. 

32# ``gemma4:12b`` is LARGE, only a bare ``gemma4`` (the 2B default) is SMALL. 

33_SMALL_MODEL_FAMILIES = ("gemma4", "gemma-4") 

34 

35# Regex to extract parameter count from a model name. 

36# 

37# Matches patterns like: 

38# :0.8b :1b :3b :7b :14b :70b — standard sizes 

39# :2.7b :1.1b :0.5b — decimal sizes 

40# model-3b-instruct model_1b_chat — common separators 

41# :8x7b :8x22b — Mixture-of-Experts (MoE) 

42# 

43# Group 1: the base parameter count (float-compatible) 

44# Group 2: if present, the MoE expert size — indicates a large MoE model 

45_PARAM_REGEX = re.compile( 

46 r"(\d+(?:\.\d+)?)(?:x(\d+))?b(?![a-z0-9])", 

47 re.IGNORECASE, 

48)

51def detect_model_size(model_name: Optional[str]) -> ModelSize: 

52 """ 

53 Detect model size by parsing the actual parameter count from the name. 

54 

55 Uses a regex to extract the parameter count (e.g. "0.8b", "7b", "14b") 

56 and compares it against the SMALL_MODEL_THRESHOLD. MoE models (e.g. 

57 "8x7b") are treated as LARGE since their total parameter count is much 

58 higher. Gemma 4 variants with a known size tag follow the same threshold; 

59 only bare Gemma 4 names with no size indicator default to SMALL. 

60 

61 Args: 

62 model_name: Model name (e.g., "qwen3.5:0.8b", "qwen3.8:27b", 

63 "mixtral:8x7b") 

64 

65 Returns: 

66 ModelSize.SMALL for models ≤7.5B parameters, ModelSize.LARGE otherwise 

67 """ 

68 if not model_name: 

69 return ModelSize.LARGE # Default to large for safety 

70 

71 name_lower = model_name.lower() 

72 

73 # Extract parameter count from model name — this runs first so that 

74 # known sizes are honoured even for Gemma 4 variants (e.g. a 12B Gemma 4 

75 # should be treated as LARGE, not SMALL). 

76 match = _PARAM_REGEX.search(name_lower) 

77 if match: 

78 # MoE format (e.g. "8x7b") — total parameters are much larger than the 

79 # base number implies, so classify as LARGE 

80 if match.group(2) is not None: 

81 return ModelSize.LARGE 

82 

83 param_count = float(match.group(1)) 

84 

85 if param_count <= _SMALL_MODEL_THRESHOLD: 

86 return ModelSize.SMALL 

87 else: 

88 return ModelSize.LARGE 

89 

90 # No size indicator found. Fall back to model-family heuristics. 

91 # Gemma 4 variants without an explicit size tag (e.g. bare "gemma4") 

92 # are assumed to be the default small variant. 

93 for family in _SMALL_MODEL_FAMILIES: 

94 if family in name_lower: 

95 return ModelSize.SMALL 

96 

97 return ModelSize.LARGE # No size indicator found

223# Explicit constraints for small models - focused specifically on the greeting case 

224# without being overly restrictive on legitimate tool use. 

225# NOTE: Repeated twice (x2) intentionally. Research shows repeating key instructions 

226# improves instruction-following in smaller models. 

227# See: "The Power of Noise: Redefining Retrieval for RAG Systems" (arXiv:2401.14887) 

228# and "Lost in the Middle: How Language Models Use Long Contexts" (arXiv:2307.03172) 

229# Repetition places the constraint both early (primacy) and late (recency) in the prompt. 

230# NB: these constraints are intentionally phrased WITHOUT language-specific 

231# examples of forbidden phrasing ("would you like me to", "I can search", etc.) 

232# because this assistant supports an arbitrary set of languages. We describe 

233# the BEHAVIOURS to avoid, not English tokens that happen to express them. 

234# Small models still get enough structure to follow because each rule is 

235# stated in imperative form with a concrete trigger + action. 

236_TOOL_CONSTRAINTS_BASE = """ACTION REQUESTS — NEVER REFUSE BEFORE CHECKING: 

237When the user asks for an action (open something, navigate somewhere, send a message, look something up, play something, fetch data), scan your available tools FIRST and call the one whose description covers that action. Do NOT apologise, do NOT say "I cannot do that", do NOT describe your limitations — just call the tool. If nothing in your current tool list obviously fits, call `toolSearchTool` with a short description of the action before giving up. A false refusal when a tool exists is the worst possible reply; calling a tool that turns out not to help is recoverable. Treat "I cannot" as a last resort reserved for when both your tool list AND `toolSearchTool` have been exhausted. 

238 

239GREETING HANDLING: 

240When the user's message is a greeting or casual social phrase (whatever language), respond directly and warmly WITHOUT calling any tools. Greetings do not require external data. 

241 

242WEATHER AND CURRENT CONDITIONS: 

243When the user brings up weather, climate, temperature, or current conditions — whether as a question ("what is the weather like"), a statement ("it is nice today", "it is cold out there"), a contextual remark ("I am in London" after a prior weather exchange), or otherwise — call getWeather immediately with no arguments. Do NOT offer an opinion, an observation, or a generic pleasantry about the weather before calling the tool. The tool provides the actual readings; your training data does not know today's temperature anywhere. Even a casual weather remark is a trigger for getWeather. 

244 

245USER INSTRUCTIONS: 

246When the user gives you instructions about how to behave or respond (units, brevity, language, tone), acknowledge and respond directly WITHOUT calling tools. These are behavioural instructions, not data requests. 

247 

248UNKNOWN NAMED ENTITIES: 

249If the user asks about a specific named thing (a film, book, song, game, product, person, company, place, event) and you do not have concrete factual information about that exact entity, call webSearch in the SAME turn — silently. Do not offer to search, do not ask permission to search, do not announce the search, do not say you have no information and stop. If the query names the entity clearly enough to search, SEARCH — do not ask the user to disambiguate first. Clarifying BEFORE a tool call is a deflection; clarifying AFTER the tool returns nothing useful is fine. 

250 

251Any phrasing that requests information about a named entity is a search trigger — the request doesn't have to contain the word "search". Treat "tell me about X", "tell me more about X", "what do you know about X", "what can you tell me about X", "have you heard of X", and their equivalents in any language as information requests about X, not as capability questions about yourself. The correct response is to look X up and answer — not to describe what you can or cannot do. 

252 

253Only skip the lookup if you can state concrete facts about the exact entity (title, year, creator, plot) without guessing. A diary or memory mention of the entity's name only confirms the topic came up — it does NOT give you facts you can state. Never invent plot, cast, release year, themes, or other specifics from prior knowledge. If you do not have facts from a tool result in this turn, you must call webSearch. 

254 

255ARGUMENTS THE TOOL CAN AUTO-DERIVE: 

256If a tool's description says it has a default for some argument (for example getWeather uses the user's current location when none is given), call the tool in the SAME turn with whatever arguments you do have — even zero — and let the tool fill the rest. Do NOT ask the user to supply that argument. Do NOT reply with a clarifying question like "which location?" or "where are you?" when the tool's description already states it auto-derives that argument. Concretely: a message like "how's the weather today" must trigger getWeather immediately with no arguments, NOT a question back to the user. Asking for an argument the tool auto-derives wastes a turn and frustrates the user. 

257 

258SELF-CONTAINED TOOL ARGUMENTS: 

259Whenever you call any tool with a free-form text argument (a search query, lookup string, question field — whatever the tool names it), the string you pass MUST be a self-contained restatement of the user's intent. Resolve pronouns, ellipsis, and implicit references from earlier turns yourself — the tool does NOT see the conversation history, it only sees the argument you pass. If the previous turn was about "Harry Styles" and the user now asks "what are his most famous songs?", the argument must be something like "Harry Styles most famous songs", NOT "what are his most famous songs". Prefer a compact keyword phrase over a conversational sentence. Never pass the user's literal utterance through when it contains unresolved pronouns, "that", "those", "it", "his", "her", "their", or similar references. This applies to every tool — webSearch, Wikipedia, MCP tools, all of them.""" 

260 

261# Repeat the constraints twice for better instruction-following in small models 

262TOOL_CONSTRAINTS_SMALL = _TOOL_CONSTRAINTS_BASE + "\n\n" + _TOOL_CONSTRAINTS_BASE
