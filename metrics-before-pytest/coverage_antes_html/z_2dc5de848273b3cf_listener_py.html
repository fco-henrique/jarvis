145def _resample(audio, src_rate: int, dst_rate: int): 

146 """Resample a 1-D float32 numpy array from *src_rate* to *dst_rate*. 

147 

148 Uses linear interpolation — fast and good enough for speech going into Whisper. 

149 """ 

150 if src_rate == dst_rate or np is None: 

151 return audio 

152 ratio = dst_rate / src_rate 

153 n_out = int(len(audio) * ratio) 

154 indices = np.arange(n_out) / ratio 

155 return np.interp(indices, np.arange(len(audio)), audio).astype(np.float32)

220 """Probe cuBLAS + cuDNN availability once per process and cache the result. 

221

223 versions in `installer/windows/install_cuda.ps1` (`cublas64_12.dll`, 

224 `cudnn_ops64_9.dll`) so a future installer bump doesn't silently fall 

225 back to CPU until this probe is updated too. A bump outside the 

226 existing range still requires widening these ranges — the relationship 

227 is by convention, not enforced. 

228 

229 Cached because DLLs don't appear or disappear while the process is

236 cudnn_found = False 

237 try: 

238 import ctypes 

239 

240 for ver in range(20, 10, -1): 

241 try: 

242 ctypes.CDLL(f"cublas64_{ver}.dll") 

243 cublas_found = True 

244 debug_log(f"cuBLAS found (cublas64_{ver}.dll)", "voice") 

245 break 

246 except OSError: 

247 continue

351def _clear_corrupted_whisper_cache(error_message: str) -> bool: 

352 """Clear a corrupted Whisper model cache directory. 

353 

354 Parses the CTranslate2 error message to find the snapshot directory, 

355 then deletes the parent ``models--`` directory so the model can be 

356 re-downloaded cleanly (including blobs that may also be corrupt). 

357 

358 Returns ``True`` if a cache directory was found and deleted. 

359 """ 

360 import re 

361 import shutil 

362 

363 # CTranslate2 error format: 

364 # "Unable to open file 'model.bin' in model '/path/to/snapshots/hash'" 

365 match = re.search( 

366 r"unable to open file\s+'[^']+'\s+in model\s+'([^']+)'", 

367 error_message, 

368 re.IGNORECASE, 

369 ) 

370 if not match: 

371 debug_log("could not parse cache path from error message", "voice") 

372 return False 

373 

374 snapshot_path = match.group(1) 

375 

376 # Walk up to the models-- directory 

377 # snapshot_path is e.g. .../models--Org--Name/snapshots/<hash> 

378 # We want to delete .../models--Org--Name entirely 

379 from pathlib import Path 

380 path = Path(snapshot_path) 

381 model_dir = None 

382 for parent in [path] + list(path.parents): 

383 if parent.name.startswith("models--"): 

384 model_dir = parent 

385 break 

386 

387 if model_dir is None or not model_dir.is_dir(): 

388 debug_log(f"could not locate models-- cache directory from: {snapshot_path}", "voice") 

389 return False 

390 

391 try: 

392 shutil.rmtree(model_dir) 

393 debug_log(f"cleared corrupted Whisper cache: {model_dir}", "voice") 

394 return True 

395 except OSError as e: 

396 debug_log(f"failed to clear corrupted cache: {e}", "voice") 

397 return False
