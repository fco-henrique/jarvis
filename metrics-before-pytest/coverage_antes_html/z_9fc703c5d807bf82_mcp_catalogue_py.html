1""" 

2🔌 Curated catalogue of popular, verified MCP servers. 

3 

4Shared between the setup wizard (quick picks) and settings window (full management). 

5Each entry contains the config needed to add the server to config.json. 

6 

7Selection criteria: 

8- Must NOT duplicate Jarvis built-in tools (web search, page fetch, file ops, 

9 memory/recall, weather, screenshot/OCR, meals). 

10- Wizard-featured entries must be zero-config (no API keys). 

11- All entries must be from the official @modelcontextprotocol org or widely trusted. 

12""" 

13 

14from __future__ import annotations 

15 

16from dataclasses import dataclass, field 

17from typing import Dict, List, Optional 

18 

19 

20@dataclass 

21class MCPEntry: 

22 """A curated MCP server entry.""" 

23 name: str # Config key / server name 

24 display_name: str # Human-readable name 

25 description: str # Short description of what it does 

26 command: str # Executable (e.g. "npx") 

27 args: List[str] # Command arguments 

28 env: Dict[str, str] = field(default_factory=dict) 

29 needs_api_key: bool = False # Requires user to supply an API key 

30 api_key_env_var: Optional[str] = None # Which env var holds the key 

31 api_key_hint: Optional[str] = None # Help text for obtaining the key 

32 wizard_featured: bool = False # Show in setup wizard quick picks 

33 category: str = "general" # Grouping for display 

34 

35 def to_config(self, extra_env: Optional[Dict[str, str]] = None) -> Dict: 

36 """Convert to the config.json MCP entry format. 

37 

38 Args: 

39 extra_env: Additional env vars to merge (e.g. user-supplied API keys). 

40 Never mutates the entry's own env dict. 

41 """ 

42 cfg: Dict = { 

43 "transport": "stdio", 

44 "command": self.command, 

45 "args": list(self.args), 

46 } 

47 merged_env = {**self.env, **(extra_env or {})} 

48 if merged_env: 

49 cfg["env"] = merged_env 

50 return cfg
