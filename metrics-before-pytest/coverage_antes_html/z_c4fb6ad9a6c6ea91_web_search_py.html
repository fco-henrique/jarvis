17# Per-fetch deadline — tight enough that a worst-case 3-way cascade fits the 

18# voice-assistant latency budget. Historical value was 8s per fetch (24s worst 

19# case); 4s keeps the cascade under 12s even if every attempt stalls. 

20_FETCH_TIMEOUT_SEC = 4.0 

21# Wall-clock cap for the entire cascade when fetches run in parallel. 

22_CASCADE_WALL_CLOCK_SEC = 8.0 

23# Hard ceiling on the whole provider chain (DDG + Brave + Wikipedia). Without 

24# this, a bad day where every provider stalls to timeout could run ~40s — 

25# intolerable for a voice assistant. Past this deadline the tool gives up and 

26# returns the honest-block envelope. 

27_TOTAL_WALL_CLOCK_SEC = 20.0 

28# Max redirects to follow manually (so we can re-validate each hop). 

29_MAX_REDIRECTS = 3 

30# Max bytes we'll pull from a single page before giving up. Caps prompt- 

31# injection surface and protects against hostile servers streaming forever. 

32_MAX_FETCH_BYTES = 512 * 1024

357def _language_script_mismatches_query(lang: str, query: str) -> bool: 

358 """Return True when `lang` expects a non-Latin script but `query` is 

359 overwhelmingly Latin letters. Used to catch Whisper language

365 return False

370# Per-request timeout for Wikipedia API calls. Smaller than the generic 

371# `_FETCH_TIMEOUT_SEC` because the helper makes up to three sequential calls 

372# (opensearch + optional fulltext + REST summary) and the whole branch must 

373# fit comfortably inside `_TOTAL_WALL_CLOCK_SEC`. The Wikimedia API typically 

374# responds in well under a second, so 4s is plenty without burning the chain 

375# budget on tail latency.

377# Floor on the per-request timeout when a deadline shrinks the budget. Below 

378# this we treat the budget as exhausted rather than firing a doomed-to-time- 

379# out request that still costs round-trip overhead. 

380_WIKIPEDIA_MIN_TIMEOUT_SEC = 0.5

383def _wikipedia_request_timeout(deadline: Optional[float]) -> Optional[float]: 

384 """Return the timeout to use for a Wikipedia request, honouring `deadline`. 

385 

386 Returns the configured per-request timeout when no deadline is supplied, 

387 a clamped remaining-budget value when a deadline is in the future, or 

388 `None` when the deadline has already passed (caller must skip the call). 

389 """ 

390 if deadline is None: 

391 return _WIKIPEDIA_REQUEST_TIMEOUT_SEC 

392 import time as _time 

393 remaining = deadline - _time.monotonic() 

394 if remaining < _WIKIPEDIA_MIN_TIMEOUT_SEC: 

395 return None 

396 return min(_WIKIPEDIA_REQUEST_TIMEOUT_SEC, remaining)
