83def _review_graph_facts( 

84 facts: list[tuple[str, str]], cfg, chat_model: str, 

85 timeout_sec: float, thinking: bool, 

86) -> list[tuple[str, str]]: 

87 """Retain original durable facts only after a complete semantic review.""" 

88 if not facts or timeout_sec <= 0: 

89 return [] 

90 content = "\n".join( 

91 f"{index}: {json.dumps(fact, ensure_ascii=False)}" 

92 for index, (_branch, fact) in enumerate(facts) 

93 ) 

94 try: 

95 response = call_llm_direct( 

96 cfg=cfg, chat_model=chat_model, 

97 system_prompt=_FACT_HYGIENE_PROMPT, user_content=content, 

98 timeout_sec=timeout_sec, thinking=thinking, temperature=0.0, 

99 max_tokens=_GRAPH_FACT_TOKEN_BUDGET, 

100 ) 

101 labels = {} 

102 for line in (response or "").splitlines(): 

103 match = re.fullmatch(r"(\d+)(?::|\s)\s*(DURABLE|TRANSIENT|INTERACTION|ADVICE)", line.strip()) 

104 if not match: 

105 raise ValueError("Invalid classification") 

106 index = int(match[1]) 

107 if index >= len(facts) or index in labels: 

108 raise ValueError("Invalid candidate index") 

109 labels[index] = match[2] 

110 if len(labels) != len(facts): 

111 raise ValueError("Incomplete classification") 

112 except Exception as error: 

113 debug_log(f"graph fact review unavailable: {type(error).__name__}", "memory") 

114 return [] 

115 retained = [fact for index, fact in enumerate(facts) if labels[index] == "DURABLE"] 

116 debug_log(f"graph fact review: retained {len(retained)} of {len(facts)} candidates", "memory") 

117 return retained

388def find_best_node( 

389 store: GraphMemoryStore, 

390 fragment: str, 

391 cfg, 

392 chat_model: str, 

393 timeout_sec: float = 15.0, 

394 thinking: bool = False, 

395 picker_model: Optional[str] = None, 

396 branch_root_id: Optional[str] = None, 

397) -> str: 

398 """Find the best node to store a memory fragment. 

399 

400 When ``branch_root_id`` is provided (one of the fixed taxonomy 

401 branches — User / Directives / World), the shortcut entry points 

402 (recent / top) are skipped entirely and traversal descends only 

403 through that branch's subtree. This guarantees the purpose-shaped 

404 top-level taxonomy is respected — a User fact can never end up in 

405 the World subtree just because a World node happened to be 

406 recently accessed. 

407 

408 When ``branch_root_id`` is None (legacy callers), the old three- 

409 entry-point heuristic is used: 

410 

411 1. Recent nodes — check if fragment fits a recently accessed node 

412 2. Top nodes — check frequently accessed domains 

413 3. Root traversal — greedy top-down descent from root 

414 

415 Returns the id of the best node. 

416 """ 

417 debug_log( 

418 f"graph traversal: placing '{fragment[:60]}...' " 

419 f"(branch={branch_root_id or 'any'})", 

420 "memory", 

421 ) 

422 

423 if branch_root_id is None: 

424 # Entry point 1: Check recent nodes 

425 recent = store.get_recent_nodes(limit=5) 

426 if recent: 426 ↛ 439line 426 didn't jump to line 439 because the condition on line 426 was always true

427 debug_log(f"graph traversal: trying {len(recent)} recent nodes: {[n.name for n in recent]}", "memory") 

428 best = _llm_pick_best_child( 

429 fragment, recent, cfg, chat_model, 

430 timeout_sec=timeout_sec, thinking=thinking, picker_model=picker_model, 

431 ) 

432 if best is not None: 

433 matched = store.get_node(best) 

434 name = matched.name if matched else best[:8] 

435 debug_log(f"graph traversal: matched recent node '{name}'", "memory") 

436 return best 

437 

438 # Entry point 2: Check top nodes (excluding any already checked as recent) 

439 recent_ids = {n.id for n in recent} if recent else set() 

440 top = [n for n in store.get_top_nodes(limit=10) if n.id not in recent_ids] 

441 if top: 

442 debug_log(f"graph traversal: trying {len(top)} top nodes: {[n.name for n in top]}", "memory") 

443 best = _llm_pick_best_child( 

444 fragment, top, cfg, chat_model, 

445 timeout_sec=timeout_sec, thinking=thinking, picker_model=picker_model, 

446 ) 

447 if best is not None: 

448 matched = store.get_node(best) 

449 name = matched.name if matched else best[:8] 

450 debug_log(f"graph traversal: matched top node '{name}'", "memory") 

451 return best 

452 

453 # Entry point 3 (or sole entry point when branch is pinned): 

454 # greedy descent from the branch root (or root when no branch). 

455 start_id = branch_root_id or "root" 

456 debug_log(f"graph traversal: descending from '{start_id}'", "memory") 

457 current_id = start_id 

458 depth = 0 

459 for depth in range(MAX_TRAVERSAL_DEPTH): 459 ↛ 478line 459 didn't jump to line 478 because the loop on line 459 didn't complete

460 children = store.get_children(current_id) 

461 if not children: 

462 debug_log(f"graph traversal: leaf node at depth {depth}", "memory") 

463 break # Leaf node — write here 

464 

465 debug_log(f"graph traversal: depth {depth}, choosing from {[c.name for c in children]}", "memory") 

466 best = _llm_pick_best_child( 

467 fragment, children, cfg, chat_model, 

468 timeout_sec=timeout_sec, thinking=thinking, picker_model=picker_model, 

469 ) 

470 if best is None: 

471 debug_log(f"graph traversal: no children fit at depth {depth}, stopping", "memory") 

472 break # None of the children fit — write to current node 

473 matched = store.get_node(best) 

474 name = matched.name if matched else best[:8] 

475 debug_log(f"graph traversal: descended into '{name}'", "memory") 

476 current_id = best 

477 

478 final = store.get_node(current_id) 

479 final_name = final.name if final else current_id[:8] 

480 debug_log(f"graph traversal: writing to '{final_name}' (depth {depth})", "memory") 

481 return current_id

585def _extract_facts_object(response: str) -> Optional[dict]: 

586 """Pull a `{"facts": [...]}` object out of an LLM response. 

587 

588 Tries direct `json.loads` first (the strict prompt + T=0 should 

589 produce clean JSON in the common case). Otherwise scans every `{` 

590 and uses ``json.JSONDecoder.raw_decode`` to consume a balanced 

591 object starting there. ``raw_decode`` handles nested braces, so a 

592 fact string containing ``{`` or ``}`` parses correctly — unlike a 

593 `[^{}]`-scoped regex which would refuse to match the outer 

594 object. Returns the first parsed object that has a list-valued 

595 ``facts`` key. 

596 """ 

597 stripped = response.strip() 

598 if stripped.startswith("{"): 

599 try: 

600 parsed = json.loads(stripped) 

601 except (json.JSONDecodeError, ValueError): 

602 parsed = None 

603 if isinstance(parsed, dict) and isinstance(parsed.get("facts"), list): 603 ↛ 607line 603 didn't jump to line 607 because the condition on line 603 was always true

604 return parsed 

605 # O(n) over the response: at most one `{` per character. Picker 

606 # responses are bounded (single rewrite, T=0), so this stays cheap. 

607 for match in re.finditer(r"\{", response): 

608 try: 

609 parsed, _ = _JSON_DECODER.raw_decode(response, match.start()) 

610 except (json.JSONDecodeError, ValueError): 

611 continue 

612 if isinstance(parsed, dict) and isinstance(parsed.get("facts"), list): 

613 return parsed 

614 return None

617def merge_node_data( 

618 store: GraphMemoryStore, 

619 node_id: str, 

620 new_facts: list[str], 

621 cfg, 

622 chat_model: str, 

623 timeout_sec: float = 20.0, 

624 thinking: bool = False, 

625 picker_model: Optional[str] = None, 

626 node: Optional[MemoryNode] = None, 

627) -> MergeResult: 

628 """Merge ``new_facts`` into ``node_id``'s data via one LLM rewrite. 

629 

630 Combines the existing node data and the queued new facts, asks the 

631 model to produce a clean, consolidated, contradiction-free fact 

632 list, and writes that back as the node's full data. This subsumes 

633 dedupe, supersession, and per-write consolidation in a single 

634 pass — the latest prompt always rewrites the node, so updated 

635 conventions propagate to existing data without a separate 

636 migration step. 

637 

638 Pass an empty ``new_facts`` list to run a self-consolidation pass 

639 on the node's existing data alone (dedupe / consolidate / prune 

640 only — no fact incorporation). The merge prompt's rules apply 

641 equally to the existing data, so the same LLM call serves both 

642 "incorporate new facts" and "tidy existing facts". 

643 

644 Hallucination guard: the cleaned rewrite is rejected if it grows 

645 beyond ``len(existing_lines) + len(new_facts) + 2`` entries. 

646 Consolidation should only ever shrink or hold; runaway growth 

647 means the model invented content. 

648 

649 Fail-open on any error (LLM failure, parse failure, empty 

650 rewrite, oversized rewrite). Caller's append path then writes the 

651 fact directly. We never let a flaky LLM erase data — a 

652 contradiction is recoverable, a silent wipe is not. 

653 

654 Pass ``node`` if the caller has already fetched it; saves a 

655 redundant SQLite read on the orchestrator's hot path. 

656 """ 

657 if node is None: 

658 node = store.get_node(node_id) 

659 if node is None: 659 ↛ 660line 659 didn't jump to line 660 because the condition on line 659 was never true

660 return MergeResult(success=False) 

661 

662 existing_lines = _split_data_lines(node.data) 

663 # Re-join from the parsed lines so the prompt body and the line 

664 # count agree byte-for-byte — `existing` was previously a separate 

665 # `.strip()` of the raw blob, which could disagree with the parsed 

666 # list on edge whitespace. 

667 existing = "\n".join(existing_lines) 

668 sanitised_new: list[str] = [f.strip() for f in new_facts if f and f.strip()] 

669 

670 if not existing_lines and not sanitised_new: 670 ↛ 672line 670 didn't jump to line 672 because the condition on line 670 was never true

671 # Nothing to do. 

672 return MergeResult(success=False) 

673 

674 if not existing_lines: 

675 # Cold start: no existing data to merge against. Caller's 

676 # append path will write each new fact verbatim. Skipping the 

677 # LLM call keeps cold-start writes cheap. 

678 return MergeResult(success=False) 

679 

680 if sanitised_new: 

681 new_facts_block = "\n".join(f"- {f}" for f in sanitised_new) 

682 user_content = ( 

683 f"CURRENT facts on the node:\n{existing}\n\n" 

684 f"NEW facts to incorporate:\n{new_facts_block}" 

685 ) 

686 else: 

687 user_content = ( 

688 f"CURRENT facts on the node (no new facts to add — " 

689 f"consolidate / dedupe / prune only):\n{existing}" 

690 ) 

691 

692 effective_model = picker_model or chat_model 

693 response = call_llm_direct( 

694 cfg=cfg, 

695 chat_model=effective_model, 

696 system_prompt=_MERGE_SYSTEM_PROMPT, 

697 user_content=user_content, 

698 timeout_sec=timeout_sec, 

699 thinking=thinking, 

700 temperature=0.0, 

701 max_tokens=300, 

702 ) 

703 

704 if not response: 

705 return MergeResult(success=False) 

706 

707 parsed = _extract_facts_object(response) 

708 if parsed is None: 

709 return MergeResult(success=False) 

710 

711 cleaned: list[str] = [] 

712 for item in parsed["facts"]: 

713 if not isinstance(item, str): 

714 continue 

715 line = item.strip() 

716 if line: 

717 cleaned.append(line) 

718 

719 # Empty rewrite is suspicious — a non-empty `existing` plus 

720 # (optional) new facts should never collapse to nothing. Treat as 

721 # failure and let the caller's append path run. 

722 if not cleaned: 

723 return MergeResult(success=False) 

724 

725 # Hallucination guard: bound the output relative to the input. 

726 # Consolidation rules can shrink or hold but should never grow 

727 # beyond `existing + new + small slack` — anything larger means 

728 # the model invented content not present in either input. 

729 max_kept = len(existing_lines) + len(sanitised_new) + _MERGE_GROWTH_SLACK 

730 if len(cleaned) > max_kept: 

731 debug_log( 

732 f"merge: rejected rewrite — {len(cleaned)} lines exceeds " 

733 f"guard cap of {max_kept}", 

734 "memory", 

735 ) 

736 return MergeResult(success=False) 

737 

738 # Identify which of the new facts actually survived the rewrite. 

739 # Uses the dedupe primitive's Unicode folding plus a tolerant 

740 # trailing-punctuation strip — picker models routinely rephrase 

741 # facts by dropping the trailing full stop ("X." → "X"), and a 

742 # strict `normalise_fact` match would then under-report every 

743 # batched flush as "0 stored". A new fact missing from the 

744 # cleaned set was consolidated out, treated as a duplicate, or 

745 # silently dropped — caller can then decide whether to skip 

746 # reporting or append-fallback. 

747 def _match_key(text: str) -> str: 

748 return normalise_fact(text).rstrip(".,;:!?")

750 cleaned_keys = {_match_key(line) for line in cleaned if line.strip()} 

751 incorporated_indices: list[int] = [] 

752 for idx, fact in enumerate(new_facts): 

753 if not fact or not fact.strip(): 753 ↛ 754line 753 didn't jump to line 754 because the condition on line 753 was never true

754 continue 

755 key = _match_key(fact) 

756 if key and key in cleaned_keys: 

757 incorporated_indices.append(idx) 

758 

759 new_data = "\n".join(cleaned) 

760 store.update_node(node_id, data=new_data) 

761 return MergeResult(success=True, incorporated_indices=incorporated_indices)

1119 path for nodes that accumulated contradictions before the 

1120 merge-on-write step landed: under merge-on-write, a node only 

1121 gets cleaned when a new related fact arrives, so backlog stays 

1122 dirty until something nudges it. Calling this op nudges 

1123 everything at once. 

1124 

1125 Yields ``(node_name, lines_before, lines_after)`` per node as the 

1126 walk progresses, so a streaming caller (e.g. an NDJSON endpoint) 

1127 can surface per-node feedback in real time on graphs with many 

1128 nodes. Fail-open: a node that fails to merge is left untouched 

1129 and reported with ``lines_after == lines_before``. 

1130 """ 

1131 # Snapshot all nodes up front so a rewrite mid-walk doesn't 

1132 # cause us to revisit or skip nodes. 

1133 all_nodes = store.get_all_nodes() 

1134 for node in all_nodes: 

1135 if not is_populated_node(node): 

1136 continue 

1137 before = len(_split_data_lines(node.data)) 

1138 try: 

1139 result = merge_node_data( 

1140 store=store, 

1141 node_id=node.id, 

1142 new_facts=[], 

1143 cfg=cfg, 

1144 chat_model=chat_model, 

1145 timeout_sec=timeout_sec, 

1146 thinking=thinking, 

1147 picker_model=picker_model, 

1148 node=node, 

1149 ) 

1150 except Exception as e: 

1151 debug_log(f"consolidate-all: failed for '{node.name}' — {e}", "memory") 

1152 result = MergeResult(success=False) 

1153 

1154 refreshed = store.get_node(node.id) 

1155 after = len(_split_data_lines(refreshed.data)) if refreshed else before 

1156 debug_log( 

1157 f"consolidate-all: '{node.name}' {before} → {after} lines " 

1158 f"(success={result.success})", 

1159 "memory", 

1160 ) 

1161 yield (node.name, before, after)

1167def _collect_branch_text( 

1168 store: GraphMemoryStore, branch_root_id: str, max_chars: int, 

1169) -> str: 

1170 """Return the concatenated ``data`` of all nodes in a branch's subtree, 

1171 newest-touched first, truncated at ``max_chars``. 

1172 

1173 Used to build the warm blob. We walk the subtree breadth-first from 

1174 the branch root so fresher / more-touched nodes (ordered by the 

1175 store's decayed access score) appear first; content gets truncated 

1176 at the char cap so the system prompt stays bounded. 

1177 """ 

1178 root = store.get_node(branch_root_id) 

1179 if root is None: 1179 ↛ 1180line 1179 didn't jump to line 1180 because the condition on line 1179 was never true

1180 return "" 

1181 

1182 parts: list[str] = [] 

1183 remaining = max_chars 

1184 # BFS ordered by sibling decayed-access score (get_children sorts). 

1185 queue: list[str] = [branch_root_id] 

1186 visited: set[str] = set() 

1187 while queue and remaining > 0: 

1188 node_id = queue.pop(0) 

1189 if node_id in visited: 1189 ↛ 1190line 1189 didn't jump to line 1190 because the condition on line 1189 was never true

1190 continue 

1191 visited.add(node_id) 

1192 node = store.get_node(node_id) 

1193 if node is None: 1193 ↛ 1194line 1193 didn't jump to line 1194 because the condition on line 1193 was never true

1194 continue 

1195 if node.data: 

1196 snippet = node.data.strip() 

1197 if len(snippet) > remaining: 

1198 snippet = snippet[: max(0, remaining - 1)].rstrip() + "…" 

1199 if snippet: 1199 ↛ 1202line 1199 didn't jump to line 1202 because the condition on line 1199 was always true

1200 parts.append(snippet) 

1201 remaining -= len(snippet) + 1 # +1 for separator 

1202 for child in store.get_children(node_id): 

1203 queue.append(child.id) 

1204 return "\n".join(parts)

1207def build_warm_profile( 

1208 store: GraphMemoryStore, 

1209 *, 

1210 user_max_chars: int = 1200, 

1211 directives_max_chars: int = 600, 

1212) -> dict[str, str]: 

1213 """Build the warm profile blob from the User and Directives branches. 

1214 

1215 Returned as a dict of ``{"user": "...", "directives": "..."}`` so 

1216 callers can render the two sections separately in the system prompt 

1217 (directives want a near-verbatim, imperative framing; user facts 

1218 want a descriptive framing). An empty string on either key means 

1219 the branch is empty — the caller should omit that section entirely, 

1220 not render an empty heading. 

1221 

1222 Call sites should cache this per-session and invalidate on writes 

1223 to the User or Directives branches, since it's injected on every 

1224 reply turn. Recomputing from the store on every turn is cheap 

1225 (SQLite reads only, no LLM calls) but still wasteful at scale. 

1226 """ 

1227 return { 

1228 "user": _collect_branch_text(store, BRANCH_USER, user_max_chars), 

1229 "directives": _collect_branch_text( 

1230 store, BRANCH_DIRECTIVES, directives_max_chars, 

1231 ), 

1232 }

1235def format_warm_profile_block(profile: dict[str, str]) -> str: 

1236 """Render a warm profile dict as a labelled block for the system prompt. 

1237 

1238 Returns an empty string when both sections are empty so the caller 

1239 can append unconditionally without introducing whitespace noise on 

1240 fresh installs with no accumulated memory. 

1241 

1242 The labels deliberately mirror the denial templates small models 

1243 produce under uncertainty ("I don't have information the user has 

1244 shared in prior conversations"). Naming the section exactly what 

1245 the denial refers to short-circuits the denial pattern — see the 

1246 CLAUDE.md note on denial-template mirroring. 

1247 """ 

1248 user = (profile.get("user") or "").strip() 

1249 directives = (profile.get("directives") or "").strip() 

1250 if not user and not directives: 

1251 return "" 

1252 

1253 sections: list[str] = [] 

1254 if user: 

1255 sections.append( 

1256 "INFORMATION THE USER HAS SHARED IN PRIOR CONVERSATIONS\n" 

1257 "(their identity, location, tastes, preferences, habits, " 

1258 "history — treat this as known context about the user, not " 

1259 "as new information you need to ask about):\n" 

1260 f"{user}" 

1261 ) 

1262 if directives: 

1263 sections.append( 

1264 "STANDING INSTRUCTIONS FROM THE USER\n" 

1265 "(rules the user has told you to follow — obey these " 

1266 "verbatim, in every reply, without being reminded):\n" 

1267 f"{directives}" 

1268 ) 

1269 return "\n\n".join(sections)
