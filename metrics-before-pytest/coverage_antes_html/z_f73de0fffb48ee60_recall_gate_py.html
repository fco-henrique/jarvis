35def _content_words(text: str) -> set[str]: 

36 # \w with UNICODE (default in Py3) matches letters in any script — 

37 # Latin, Cyrillic, CJK, Arabic, Hebrew, etc. Keeps Jarvis language-agnostic 

38 # per CLAUDE.md. Digit-only runs are excluded by the stopword-style filter. 

39 words = re.findall(r"\w{3,}", (text or "").lower(), flags=re.UNICODE) 

40 return {w for w in words if w not in _STOPWORDS and not w.isdigit()}

48def should_recall( 

49 query: str, 

50 recent_messages: List[dict], 

51 *, 

52 min_coverage: float = 0.5, 

53) -> bool: 

54 """Return True iff diary/graph recall should run for this query. 

55 

56 False only when: 

57 1. Hot-window contains at least one fresh tool result, AND 

58 2. At least `min_coverage` fraction of the query's content words 

59 appear in the combined hot-window text (coverage, not symmetric 

60 Jaccard — the window is always larger than the query). 

61 

62 Fail-open: any exception or missing data → True. 

63 """ 

64 try: 

65 if not recent_messages: 

66 return True 

67 if not _has_fresh_tool_result(recent_messages): 

68 return True 

69 q_words = _content_words(query) 

70 if not q_words: 

71 # Stopword-only query cannot justify skipping recall. 

72 return True 

73 window_text_parts: list[str] = [] 

74 for m in recent_messages: 

75 c = m.get("content") 

76 if isinstance(c, str) and c: 

77 window_text_parts.append(c) 

78 window_words = _content_words(" ".join(window_text_parts)) 

79 if not window_words: 79 ↛ 80line 79 didn't jump to line 80 because the condition on line 79 was never true

80 return True 

81 overlap = q_words & window_words 

82 coverage = len(overlap) / len(q_words) if q_words else 0.0 

83 if coverage >= min_coverage: 

84 # Overlap words come from the user query and may carry names or 

85 # PII; push them through the structural scrub before logging so 

86 # debug logs don't become a side-channel. 

87 safe_overlap = redact(" ".join(sorted(overlap)[:5])) 

88 debug_log( 

89 f"recall gate: skip (coverage={coverage:.2f}, overlap=[{safe_overlap}])", 

90 "memory", 

91 ) 

92 return False 

93 return True 

94 except Exception as e: 

95 debug_log(f"recall gate failed open: {e}", "memory") 

96 return True
