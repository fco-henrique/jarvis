1""" 

2Compound-query decomposition helper. 

3 

4Small models (text-based tool calling) struggle to multi-step when a user asks 

5two questions joined by a conjunction — they answer one side and stop. The 

6engine splits such queries upfront so it can inject a targeted "still 

7unanswered" nudge after each tool result. 

8 

9Language-aware: conjunction shape varies wildly across languages (whitespace 

10boundaries for Latin/Cyrillic, character-level for CJK, enclitic particles 

11for Arabic/Hebrew that can't be split on safely). We keep a small per- 

12language rule table and fall back to "no decomposition" when the language 

13is unknown, rather than misapplying rules from a different family. 

14""" 

15 

16from __future__ import annotations 

17 

18import re 

19from dataclasses import dataclass 

20from typing import Optional 

21 

22# Minimum length of EACH sub-clause after the split. Empirical default tuned 

23# against ``evals/test_complex_flows.py::TestMultiStepEntityQuery`` — filters 

24# out short idiomatic phrases (English "rock and roll", French "va et vient", 

25# German "hin und her") without dropping typical multi-part entity queries 

26# whose clauses usually exceed 15 characters each. CJK languages use a 

27# smaller threshold (see ``_RULES``) because each character carries far more 

28# semantic weight than a Latin letter. 

29DEFAULT_MIN_CLAUSE_CHARS = 9 

30CJK_MIN_CLAUSE_CHARS = 4 

31# Back-compat alias kept for existing tests that imported the original constant. 

32MIN_CLAUSE_CHARS = DEFAULT_MIN_CLAUSE_CHARS 

33 

34 

35@dataclass(frozen=True) 

36class _LangRule: 

37 """Splitting policy for one language. 

38 

39 ``pattern`` matches the conjunction boundary. For languages that use 

40 whitespace between words the pattern includes ``\\s+`` padding; for CJK 

41 it matches the conjunction character(s) directly so "电影和音乐" splits 

42 cleanly without requiring authors to insert spaces. 

43 """ 

44 pattern: re.Pattern[str] 

45 min_clause_chars: int = DEFAULT_MIN_CLAUSE_CHARS 

46 

47 

48def _ws(words: str) -> re.Pattern[str]: 

49 """Whitespace-bounded conjunction pattern, case-insensitive.""" 

50 return re.compile(rf"\s+(?:{words})\s+", flags=re.IGNORECASE) 

51 

52 

53# Per-language rules. Only languages we can reasonably vouch for — either 

54# structurally (whitespace-separated families where the pattern is 

55# mechanical) or with explicit testing (see ``tests/test_compound_query.py``). 

56# Languages outside this table fall through to "no decomposition" rather 

57# than risk mis-splitting with borrowed rules. 

58_RULES: dict[str, _LangRule] = { 

59 # ── Germanic / Romance (whitespace-separated) ───────────────────────── 

60 "en": _LangRule(_ws("and")), 

61 "es": _LangRule(_ws("y|e")), # "e" before i-/hi- words 

62 "fr": _LangRule(_ws("et")), 

63 "de": _LangRule(_ws("und")), 

64 "pt": _LangRule(_ws("e")), 

65 "it": _LangRule(_ws("e|ed")), # "ed" before vowel 

66 "nl": _LangRule(_ws("en")), 

67 "sv": _LangRule(_ws("och")), 

68 "no": _LangRule(_ws("og")), # Norwegian (Bokmål) 

69 "da": _LangRule(_ws("og")), # Danish 

70 "fi": _LangRule(_ws("ja|sekä")), # Finnish 

71 # ── Slavic (Cyrillic + Latin) ───────────────────────────────────────── 

72 "ru": _LangRule(_ws("и|а также")), 

73 "uk": _LangRule(_ws("і|та|й")), # Ukrainian — і / та / й 

74 "be": _LangRule(_ws("і|ды")), # Belarusian 

75 "pl": _LangRule(_ws("i|oraz")), 

76 "cs": _LangRule(_ws("a|i")), # Czech 

77 "sk": _LangRule(_ws("a|i")), # Slovak 

78 "bg": _LangRule(_ws("и")), # Bulgarian 

79 "sr": _LangRule(_ws("и|i")), # Serbian (both scripts) 

80 "hr": _LangRule(_ws("i")), # Croatian 

81 "sl": _LangRule(_ws("in")), # Slovenian 

82 # ── Other European ──────────────────────────────────────────────────── 

83 "el": _LangRule(_ws("και|κι")), # Greek 

84 "tr": _LangRule(_ws("ve")), 

85 "hu": _LangRule(_ws("és|meg")), # Hungarian 

86 "ro": _LangRule(_ws("și|şi")), # Romanian (both diacritics) 

87 # ── Asian (whitespace-separated) ────────────────────────────────────── 

88 "vi": _LangRule(_ws("và")), # Vietnamese 

89 "id": _LangRule(_ws("dan")), # Indonesian 

90 "ms": _LangRule(_ws("dan")), # Malay 

91 "hi": _LangRule(_ws("और|तथा")), # Hindi (Devanagari) 

92 # ── CJK (no whitespace around conjunctions) ─────────────────────────── 

93 # Chinese: 和 / 与 / 以及 / 并且 — common coordinating conjunctions. 

94 # Pattern matches either a character-level conjunction OR the two-char 

95 # forms. Clause-length threshold is lowered to CJK_MIN_CLAUSE_CHARS 

96 # because each Han character carries word-level meaning. 

97 "zh": _LangRule( 

98 re.compile(r"以及|并且|以及|和|与"), 

99 min_clause_chars=CJK_MIN_CLAUSE_CHARS, 

100 ), 

101 # Japanese: そして / および / また are freestanding sentence-level 

102 # connectors. We intentionally avoid the enclitic particles と/や — 

103 # they attach to nouns and splitting on them produces nonsense. Users 

104 # who write multi-part questions typically use the freestanding forms. 

105 "ja": _LangRule( 

106 re.compile(r"そして|および|また|かつ"), 

107 min_clause_chars=CJK_MIN_CLAUSE_CHARS, 

108 ), 

109 # Korean: 그리고 / 및 are freestanding; 와/과 are postpositional 

110 # particles attached to the preceding noun, so we avoid those for the 

111 # same reason as Japanese. Allow optional whitespace around the 

112 # freestanding forms since Korean usage varies. 

113 "ko": _LangRule( 

114 re.compile(r"\s*(?:그리고|및)\s*"), 

115 min_clause_chars=CJK_MIN_CLAUSE_CHARS, 

116 ), 

117} 

118# Languages NOT included on purpose: 

119# - Arabic (ar) / Hebrew (he): the conjunction "و" / "ו" is an enclitic 

120# prefix attached directly to the following word (e.g. "وكتاب" = "and a 

121# book"). A safe split would need a morphological tokenizer; a regex 

122# produces silent false positives on every word starting with "و"/"ו". 

123# - Thai (th), Khmer (km), Lao (lo): no inter-word whitespace and the 

124# conjunctions overlap common syllables; same tokenizer requirement as 

125# above, without a cheap workaround. 

126 

127 

128def _normalise_language(language: Optional[str]) -> Optional[str]: 

129 """Return a lowercase ISO-639-1 code or None for unknown input. 

130 

131 Accepts locale-style codes like "en-US" or "zh-CN" and returns the 

132 primary subtag. Returns None for empty strings, non-strings, or 

133 tags whose primary subtag is not a valid ISO-639-1 alpha-2 code. 

134 """ 

135 if not language or not isinstance(language, str): 

136 return None 

137 code = language.strip().lower().split("-")[0][:2] 

138 return code if code.isalpha() and len(code) == 2 else None 

139 

140 

141def split_compound_query(text: str, language: Optional[str] = None) -> list[str]: 

142 """Split a compound question into ordered sub-questions. 

143 

144 Returns an empty list when the query is not compound, the language is 

145 unknown/unsupported, or either clause is shorter than the language's 

146 minimum clause length. Callers should treat an empty list as "run the 

147 query as a single unit" — we never guess across languages we don't 

148 explicitly support. 

149 """ 

150 if not text or not isinstance(text, str): 

151 return [] 

152 

153 # Default to English when language is not provided (non-voice entrypoints 

154 # like evals and text chat carry no ISO code). Voice flows always pass a 

155 # Whisper-detected language; if that language isn't in our table, we 

156 # return no decomposition rather than fall back to English and mis-split. 

157 code = _normalise_language(language) or "en" 

158 rule = _RULES.get(code) 

159 if rule is None: 

160 return [] 

161 

162 parts = rule.pattern.split(text, maxsplit=1) 

163 if len(parts) != 2: 

164 return [] 

165 

166 left, right = parts[0].strip(), parts[1].strip() 

167 if len(left) < rule.min_clause_chars or len(right) < rule.min_clause_chars: 

168 return [] 

169 return [left, right]
